"use client";

import { useEffect, useState } from "react";
import Link from "next/link";
import { useParams } from "next/navigation";
import { api } from "@/lib/api";

export default function CircuitDetailPage() {
  const params = useParams();
  const id = params.id;

  const [circuit, setCircuit] = useState(null);
  const [places, setPlaces] = useState([]);
  const [hotels, setHotels] = useState([]);

  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState("");

  const [form, setForm] = useState({
    title: "",
    description: "",
    price: "",
  });

  async function loadData() {
    try {
      setLoading(true);
      setError("");

      const [
        circuitData,
        placesData,
        hotelsData,
      ] = await Promise.all([
        api.get(`/circuits/${id}`),
        api.get("/tourist-places"),
        api.get("/hotels"),
      ]);

      setCircuit(circuitData);
      setPlaces(placesData || []);
      setHotels(hotelsData || []);

      setForm({
        title: circuitData.title || "",
        description:
          circuitData.description || "",
        price: circuitData.price || "",
      });
    } catch (err) {
      setError(
        err.message ||
          "Impossible de charger le circuit."
      );
    } finally {
      setLoading(false);
    }
  }

  useEffect(() => {
    if (id) {
      loadData();
    }
  }, [id]);

  async function handleSave(e) {
    e.preventDefault();

    try {
      setSaving(true);
      setError("");

      await api.put(`/circuits/${id}`, {
        title: form.title.trim(),
        description:
          form.description.trim(),
        price: Number(form.price),
      });

      await loadData();
    } catch (err) {
      setError(err.message);
    } finally {
      setSaving(false);
    }
  }

  async function addPlace(placeId) {
    try {
      await api.post(
        `/circuits/${id}/places/${placeId}`,
        {}
      );

      await loadData();
    } catch (err) {
      setError(err.message);
    }
  }

  async function removePlace(placeId) {
    try {
      await api.del(
        `/circuits/${id}/places/${placeId}`
      );

      await loadData();
    } catch (err) {
      setError(err.message);
    }
  }

  async function addHotel(hotelId) {
    try {
      await api.post(
        `/circuits/${id}/hotels/${hotelId}`,
        {}
      );

      await loadData();
    } catch (err) {
      setError(err.message);
    }
  }

  async function removeHotel(hotelId) {
    try {
      await api.del(
        `/circuits/${id}/hotels/${hotelId}`
      );

      await loadData();
    } catch (err) {
      setError(err.message);
    }
  }

  if (loading) {
    return (
      <div className="py-10 text-center">
        Chargement...
      </div>
    );
  }

  if (!circuit) {
    return (
      <div className="py-10 text-center">
        Circuit introuvable.
      </div>
    );
  }

  const linkedPlaces =
    circuit.circuit_places || [];

  const linkedHotels =
    circuit.circuit_hotels || [];

  const linkedPlaceIds =
    linkedPlaces.map(
      (item) => item.tourist_places?.id
    );

  const linkedHotelIds =
    linkedHotels.map(
      (item) => item.hotels?.id

    try {
      setLoading(true);
      setError("");

        api.get("/circuits"),
        api.get("/tourist-places"),
        api.get("/hotels"),
      ]);

      setCircuits(circuitsData || []);
      setPlaces(placesData || []);
      setHotels(hotelsData || []);

      setError(err.message || "Impossible de charger les données.");

      setLoading(false);
    }
  }

  useEffect(() => {
    loadData();
  }, []);

  function handleChange(e) {
    setForm({
      ...form,
      [e.target.name]: e.target.value,
    });
  }

  async function handleCreate(e) {
    e.preventDefault();

    if (!form.title.trim()) {
      setError("Le titre du circuit est obligatoire.");
      return;

    try {
      setCreating(true);
      setError("");

      await api.post("/circuits", {
        title: form.title.trim(),

"use client";

import { useEffect, useState } from "react";
import { api } from "@/lib/api";

export default function AdminCommentsPage() {
  const [comments, setComments] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  async function loadComments() {
    try {
      setLoading(true);
      setError("");

      const data =
        await api.get("/comments/admin");

      setComments(data || []);
    } catch (err) {
      setError(
        err.message ||
          "Impossible de charger les commentaires."
      );
    } finally {
      setLoading(false);
    }
  }

  useEffect(() => {
    loadComments();
  }, []);

  async function updateStatus(id, status) {
    try {
      await api.put(
        `/comments/${id}/status`,
        { status }
      );

      setComments((current) =>
        current.map((comment) =>
          comment.id === id
            ? { ...comment, status }
            : comment
        )
      );
    } catch (err) {
      setError(err.message);
    }
  }

  async function remove(id) {
    if (
      !window.confirm(
        "Supprimer ce commentaire ?"
      )
    ) {
      return;
    }

    try {
      await api.del(`/comments/${id}`);

      setComments((current) =>
        current.filter(
          (comment) => comment.id !== id
        )
      );
    } catch (err) {
      setError(err.message);
    }
  }

  return (
    <div className="mx-auto max-w-7xl">
      <div className="mb-6">
        <h1 className="text-2xl font-bold sm:text-3xl">
          Commentaires
        </h1>

        <p className="mt-1 text-sm text-gray-500">
          Modération des avis des utilisateurs.
        </p>
      </div>

      {error && (
        <div className="mb-6 rounded-xl border border-red-200 bg-red-50 p-4 text-sm text-red-700">
          {error}
        </div>
      )}

      <section className="rounded-2xl border bg-white p-4 shadow-sm sm:p-6">
        {loading ? (
          <p className="py-10 text-center text-sm text-gray-500">
            Chargement...
          </p>
        ) : comments.length === 0 ? (
          <p className="py-10 text-center text-sm text-gray-500">
            Aucun commentaire.
          </p>
        ) : (
          <div className="space-y-4">
            {comments.map((comment) => (
              <article
                key={comment.id}
                className="rounded-xl border p-4"
              >
                <div className="flex flex-col gap-3 sm:flex-row sm:items-start sm:justify-between">
                  <div>
                    <div className="flex items-center gap-2">
                      <span className="font-semibold">
                        {"★".repeat(
                          comment.rating || 0
                        )}
                      </span>

                      <span className="text-xs text-gray-500">
                        {comment.status}
                      </span>
                    </div>

                    <p className="mt-2 text-sm text-gray-700">
                      {comment.content}
                    </p>

                    <p className="mt-2 text-xs text-gray-500">
                      {comment.tourist_places?.name ||
                        comment.circuits?.title ||
                        "Cible inconnue"}
                    </p>
                  </div>

                  <div className="flex flex-wrap gap-2">
                    {comment.status !==

                      <button

                          )
                        }

function PhotoCarousel({ place, media }) {
  const [activeIndex, setActiveIndex] = useState(0);
  const carouselRef = useRef(null);

  if (!media || media.length === 0) {
    return (
      <div className="relative w-full h-[420px] bg-paper rounded-lg flex items-center justify-center overflow-hidden">
        <div className="text-center text-ink/40">
          <div className="text-4xl mb-2">📷</div>
          <p className="text-sm">Pas encore de photo</p>
        </div>
      </div>
    );
  }

  function handleScroll() {
    const carousel = carouselRef.current;

    if (!carousel) return;

    const slideWidth = carousel.clientWidth;
    const index = Math.round(
      carousel.scrollLeft / slideWidth
    );

    setActiveIndex(index);
  }

  return (
    <div className="space-y-3">
      <div
        ref={carouselRef}
        onScroll={handleScroll}
        className="flex gap-2 overflow-x-auto snap-x snap-mandatory rounded-lg scrollbar-hide scroll-smooth"
      >
        {media.map((item, index) => {
          const caption =
            index === 0
              ? place.name
              : item.caption || "";

          return (
            <div
              key={item.id}
              className="relative snap-center snap-always flex-none w-full"
            >
              {item.type === "photo" ? (
                <img
                  src={item.url}
                  alt={item.alt_text || place.name}
                  className="w-full h-[420px] object-cover rounded-lg"
                />
              ) : (
                <video
                  src={item.url}
                  className="w-full h-[420px] object-cover rounded-lg"
                  controls

import { supabase } from "./supabase";

const API_URL =
  process.env.NEXT_PUBLIC_API_URL || "http://localhost:5000/api";

async function request(path, options = {}) {
  const isFormData = options.body instanceof FormData;

  const {
    data: { session },
  } = await supabase.auth.getSession();

  const headers = new Headers(options.headers || {});

  if (!isFormData) {
    headers.set("Content-Type", "application/json");
  }

  if (session?.access_token) {
    headers.set(
      "Authorization",
      `Bearer ${session.access_token}`
    );
  }

  const res = await fetch(`${API_URL}${path}`, {
    ...options,
    headers,
  });

  if (!res.ok) {
    let message = `Erreur ${res.status}`;

    try {
      const data = await res.json();
      message = data.message || message;
    } catch {
      // pas de corps JSON
    }

    throw new Error(message);
  }

  if (res.status === 204) return null;

  return res.json();
}

export const api = {
  get: (path) => request(path),

  post: (path, body) =>
    request(path, {
      method: "POST",
      body:
        body instanceof FormData
          ? body
          : JSON.stringify(body),
    }),

  put: (path, body) =>
    request(path, {
      method: "PUT",
      body: JSON.stringify(body),
    }),

  del: (path) =>
    request(path, {
      method: "DELETE",
    }),
};
