-- ============================================================
-- Phase 1b — accès d'écriture pour le prototype sans authentification
-- ============================================================
-- Le frontend appelle actuellement l'API Express sans compte utilisateur.
-- On autorise donc temporairement les opérations CRUD via le rôle anon.
-- À la Phase 2, ces policies devront être remplacées par des policies
-- basées sur auth.uid() et les rôles administrateur.

DO $$
DECLARE
  t text;
BEGIN
  FOREACH t IN ARRAY ARRAY['tourist_places', 'circuits', 'hotels', 'media', 'circuit_places', 'circuit_hotels']
  LOOP
    EXECUTE format('ALTER TABLE public.%I ENABLE ROW LEVEL SECURITY;', t);

    IF NOT EXISTS (
      SELECT 1 FROM pg_policies
      WHERE schemaname = 'public' AND tablename = t AND policyname = 'Prototype anon insert'
    ) THEN
      EXECUTE format('CREATE POLICY "Prototype anon insert" ON public.%I FOR INSERT TO anon WITH CHECK (true);', t);
    END IF;

    IF NOT EXISTS (
      SELECT 1 FROM pg_policies
      WHERE schemaname = 'public' AND tablename = t AND policyname = 'Prototype anon update'
    ) THEN
      EXECUTE format('CREATE POLICY "Prototype anon update" ON public.%I FOR UPDATE TO anon USING (true) WITH CHECK (true);', t);
    END IF;

    IF NOT EXISTS (
      SELECT 1 FROM pg_policies
      WHERE schemaname = 'public' AND tablename = t AND policyname = 'Prototype anon delete'
    ) THEN
      EXECUTE format('CREATE POLICY "Prototype anon delete" ON public.%I FOR DELETE TO anon USING (true);', t);
    END IF;
  END LOOP;
END $$;

-- Supabase Storage : le bucket "media" doit être créé au préalable.
-- Ces policies permettent au prototype de téléverser/supprimer des fichiers
-- sans authentification. Elles devront aussi être durcies avec auth.uid()
-- lorsque l'authentification sera ajoutée.
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_policies
    WHERE schemaname = 'storage' AND tablename = 'objects' AND policyname = 'Prototype anon media select'
  ) THEN
    CREATE POLICY "Prototype anon media select"
      ON storage.objects FOR SELECT TO anon
      USING (bucket_id = 'media');
  END IF;

  IF NOT EXISTS (
    SELECT 1 FROM pg_policies
    WHERE schemaname = 'storage' AND tablename = 'objects' AND policyname = 'Prototype anon media insert'
  ) THEN
    CREATE POLICY "Prototype anon media insert"
      ON storage.objects FOR INSERT TO anon
      WITH CHECK (bucket_id = 'media');
  END IF;

  IF NOT EXISTS (
    SELECT 1 FROM pg_policies
    WHERE schemaname = 'storage' AND tablename = 'objects' AND policyname = 'Prototype anon media update'
  ) THEN
    CREATE POLICY "Prototype anon media update"
      ON storage.objects FOR UPDATE TO anon
      USING (bucket_id = 'media')
      WITH CHECK (bucket_id = 'media');
  END IF;

  IF NOT EXISTS (
    SELECT 1 FROM pg_policies
    WHERE schemaname = 'storage' AND tablename = 'objects' AND policyname = 'Prototype anon media delete'
  ) THEN
    CREATE POLICY "Prototype anon media delete"
      ON storage.objects FOR DELETE TO anon
      USING (bucket_id = 'media');

END $$;

-- ============================================================
-- Phase 1 — migrations à exécuter dans Supabase SQL Editor
-- Ne recrée aucune des 12 tables existantes, ne supprime rien.
-- Exécute ce fichier de haut en bas, en une fois.
-- ============================================================

-- 1. Table de jonction Circuit <-> TouristPlace
--    (la relation "contient" du diagramme n'était portée par aucune
--    des 12 tables ; ceci est la réalisation technique de cette relation,
--    pas une nouvelle classe UML)
create table if not exists circuit_places (
  circuit_id uuid not null references circuits(id) on delete cascade,
  tourist_place_id uuid not null references tourist_places(id) on delete restrict,
  primary key (circuit_id, tourist_place_id)
);

-- 2. Table de jonction Circuit <-> Hotel
create table if not exists circuit_hotels (
  circuit_id uuid not null references circuits(id) on delete cascade,
  hotel_id uuid not null references hotels(id) on delete restrict,
  primary key (circuit_id, hotel_id)
);

-- 3. Colonnes manquantes sur media pour la relation polymorphe
--    (nécessaires pour savoir à quoi un média est rattaché)
alter table media add column if not exists entity_type text;
alter table media add column if not exists entity_id uuid;

do $$
begin
  if not exists (
    select 1 from pg_constraint where conname = 'media_entity_type_check'
  ) then
    alter table media add constraint media_entity_type_check
      check (entity_type in ('circuit', 'tourist_place', 'hotel'));
  end if;
end $$;

-- 4. Activer le RLS sur les deux nouvelles tables
alter table circuit_places enable row level security;
alter table circuit_hotels enable row level security;

-- 5. Policies de lecture publique — le feed doit être consultable
--    sans authentification, y compris ces relations.
--    PostgreSQL ne supporte pas "CREATE POLICY IF NOT EXISTS", donc on
--    vérifie manuellement (utile car "circuits" a déjà sa policy).
do $$
declare
  t record;
begin
  for t in
    select * from (values
      ('circuits', 'Public can read circuits'),
      ('tourist_places', 'Public can read tourist_places'),
      ('hotels', 'Public can read hotels'),
      ('media', 'Public can read media'),
      ('circuit_places', 'Public can read circuit_places'),
      ('circuit_hotels', 'Public can read circuit_hotels'),
      ('languages', 'Public can read languages'),
      ('content_translations', 'Public can read content_translations')
    ) as x(tablename, policyname)
  loop
    if not exists (
      select 1 from pg_policies
      where schemaname = 'public' and tablename = t.tablename and policyname = t.policyname
    ) then
      execute format(
        'create policy %I on %I for select using (true);',
        t.policyname, t.tablename
      );
    end if;
  end loop;
end $$;

export function createAuthenticatedSupabase(token: string) {
  return createClient(supabaseUrl!, supabaseKey!, {
    global: {
      headers: {
        Authorization: `Bearer ${token}`,

import { Request, Response } from "express";
import * as circuitService from "../services/circuit.service";
import { createAuthenticatedSupabase } from "../config/supabaseAuth";

function getAuthenticatedClient(req: Request) {
  const authorization = req.headers.authorization;

  if (!authorization?.startsWith("Bearer ")) {
    throw new Error("Token d'authentification manquant.");
  }

  const token = authorization.replace("Bearer ", "");

  return createAuthenticatedSupabase(token);
}

export async function getAll(_req: Request, res: Response) {
  const circuits = await circuitService.listCircuits();
  res.json(circuits);
}

export async function getOne(req: Request, res: Response) {
  const circuit = await circuitService.getCircuitById(
    String(req.params.id)
  );

  res.json(circuit);
}

export async function create(req: Request, res: Response) {
  const client = getAuthenticatedClient(req);

  const circuit = await circuitService.createCircuit(
    client,
    req.body
  );

  res.status(201).json(circuit);
}

import { Request, Response } from "express";
import { createAuthenticatedSupabase } from "../config/supabaseAuth";
import * as commentService from "../services/comment.service";
import { AppError } from "../middlewares/errorHandler";

function getToken(req: Request) {
  const authorization = req.headers.authorization;

  if (!authorization?.startsWith("Bearer ")) {
    throw new AppError(
      "Token d'authentification manquant.",
      401
    );
  }

  return authorization.replace("Bearer ", "");
}

export async function getAll(req: Request, res: Response) {
  const touristPlaceId = req.query.tourist_place_id
    ? String(req.query.tourist_place_id)
    : undefined;

  const circuitId = req.query.circuit_id
    ? String(req.query.circuit_id)
    : undefined;

  const comments =
    await commentService.listApprovedComments(
      touristPlaceId,
      circuitId
    );

  res.json(comments);
}

// MODIFIÉ : utilise le client authentifié pour que la RLS
// (policy admin) laisse voir TOUS les commentaires, pas seulement les approuvés.
export async function getAdminAll(
  req: Request,
  res: Response
) {
  const client = createAuthenticatedSupabase(
    getToken(req)
  );

  const comments = await commentService.listComments(client);
  res.json(comments);
}

export async function create(

}

import { Request, Response, NextFunction } from "express";
import { supabase } from "../config/supabase";

export async function authenticate(
  req: Request,
  res: Response,
  next: NextFunction
) {
  try {
    const authorization = req.headers.authorization;

    if (!authorization?.startsWith("Bearer ")) {
      return res.status(401).json({
        message: "Token d'authentification manquant.",
      });
    }

    const token = authorization.replace("Bearer ", "");

    const {
      data: { user },
      error,
    } = await supabase.auth.getUser(token);

    if (error || !user) {
      return res.status(401).json({
        message: "Token d'authentification invalide.",
      });
    }

    req.user = user;

    next();
  } catch (error) {
    next(error);
  }
}

import { Request, Response, NextFunction } from "express";
import { ZodSchema } from "zod";
import { AppError } from "./errorHandler";

export function validateBody(schema: ZodSchema) {
  return (req: Request, _res: Response, next: NextFunction) => {
    const result = schema.safeParse(req.body);

    if (!result.success) {
      const message = result.error.issues
        .map((issue) => `${issue.path.join(".")} : ${issue.message}`)
        .join(" | ");
      return next(new AppError(`Données invalides — ${message}`, 422));
    }

    req.body = result.data;
    next();
  };
}

import {
  searchWikipedia,
  getWikipediaSummary,
  getWikipediaImages,
  WikiImage,
} from "./wikipedia.service";

import { searchNominatim } from "./nominatim.service";

function normalizeName(value: string): string {
  return value
    .toLowerCase()
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .replace(/\s+/g, " ")
    .trim();
}

export async function searchPlace(query: string) {
  const [wikipedia, osm] = await Promise.all([
    searchWikipedia(query),
    searchNominatim(query),
  ]);

  // Garde uniquement les résultats situés à Madagascar.
  // Et on privilégie les résultats touristiques.
  const uniqueOsm = osm
    .filter(
      (place) =>
        place.address?.country_code === "mg"
    )
    .filter((place, index, array) => {
      const normalizedCurrent = normalizeName(
        place.name || place.display_name
      );

      const duplicateIndex = array.findIndex((other) => {
        const normalizedOther = normalizeName(
          other.name || other.display_name
        );

        return normalizedCurrent === normalizedOther;
      });

      if (duplicateIndex === index) {
        return true;
      }

      const duplicate = array[duplicateIndex];

      if (
        duplicate.category === "tourism" &&
        place.category !== "tourism"
      ) {
        return false;
      }

      if (
        place.category === "tourism" &&
        duplicate.category !== "tourism"
      ) {
        return true;
      }

      return false;
    });

  const results = await Promise.all(
    uniqueOsm.map(async (place) => {
      const placeName = normalizeName(
        place.name || place.display_name
      );

      // Wikipédia uniquement si le nom correspond
      // exactement après normalisation.
      const wikiMatch = wikipedia.find((wiki) => {
        const wikiName = normalizeName(wiki.title);

        return wikiName === placeName;
      });

      let wikipediaSummary = null;
      let wikipediaImages: WikiImage[] = [];

      if (wikiMatch) {
        wikipediaSummary = await getWikipediaSummary(
          wikiMatch.title
        );

        wikipediaImages = await getWikipediaImages(
          wikiMatch.title
        );
      }

      return {
        name: place.name || place.display_name,
        latitude: Number(place.lat),
        longitude: Number(place.lon),
        address: place.display_name,
        type: place.type,
        category: place.category,
        osmId: place.osm_id,
        osmType: place.osm_type,
        wikipedia: wikipediaSummary,
        images: wikipediaImages,
      };
    })
  );

  return {
    query,
    results,
  };
}

  title: string;
  descriptionUrl: string | null;
  author: string | null;

export async function getWikipediaImages(
  title: string
): Promise<WikiImage[]> {
  const url =
    `https://${WIKI_LANG}.wikipedia.org/w/api.php?` +
    new URLSearchParams({
      action: "query",
      prop: "images",
      titles: title,
      imlimit: "10",
      format: "json",
      origin: "*",
    }).toString();

  const res = await fetch(url, {
    headers: {
      "User-Agent": USER_AGENT,
      Accept: "application/json",
    },
  });

  if (!res.ok) {
    throw new AppError(
      "Impossible de récupérer les images Wikipédia.",
      502
    );
  }

  const data = (await res.json()) as {
    query?: {
      pages?: Record<
        string,
        {
          images?: Array<{
            title: string;
          }>;
        }
      >;
    };
  };

  const pages = data.query?.pages || {};
  const page = Object.values(pages)[0];
  const images = page?.images || [];

  // On garde uniquement les formats d'images classiques.
  // Cela élimine notamment les SVG utilisés pour les icônes,
  // logos techniques, boutons, etc.
  const allowedExtensions = [
    ".jpg",
    ".jpeg",
    ".png",
    ".webp",
  ];

  const photoImages = images.filter((image) => {
    const imageTitle = image.title.toLowerCase();

    return allowedExtensions.some((extension) =>
      imageTitle.endsWith(extension)
    );
  });

  const results = await Promise.all(
    photoImages.map(async (image) => {
      const infoUrl =
        `https://${WIKI_LANG}.wikipedia.org/w/api.php?` +
        new URLSearchParams({
          action: "query",
          prop: "imageinfo",
          titles: image.title,
          iiprop: "url|extmetadata",
          iiurlwidth: "1200",
          format: "json",
          origin: "*",
        }).toString();

      const infoRes = await fetch(infoUrl, {
        headers: {
          "User-Agent": USER_AGENT,
          Accept: "application/json",
        },
      });

      if (!infoRes.ok) {
        return null;
      }

      const infoData = (await infoRes.json()) as {
        query?: {
          pages?: Record<
            string,
            {
              imageinfo?: Array<{
                thumburl?: string;
                url?: string;
                descriptionurl?: string;
                extmetadata?: {
                  Artist?: {
                    value?: string;
                  };
                  LicenseShortName?: {
                    value?: string;
                  };
                };
              }>;
            }
          >;
        };
      };

      const infoPages = infoData.query?.pages || {};
      const infoPage = Object.values(infoPages)[0];
      const info = infoPage?.imageinfo?.[0];

      if (!info) {
        return null;
      }

      return {
        url: info.thumburl || info.url || "",
        title: image.title.replace(/^File:/, ""),
        descriptionUrl: info.descriptionurl || null,
        author:
          info.extmetadata?.Artist?.value || null,
        license:
          info.extmetadata?.LicenseShortName?.value || null,
      };
    })
  );

  return results.filter(
    (image): image is WikiImage =>
      image !== null && image.url.length > 0
  );
}

# Frontend Phase 1 — interface de test pour le backend Phase 1

Ce n'est pas un prototype jetable : c'est le vrai début de ton frontend

pas d'authentification) parce que le backend n'en a pas encore. Le
bandeau en haut de l'app le rappelle.

## 1. Copier les fichiers

Copie `app/`, `components/`, `lib/` dans ton projet Next.js déjà
initialisé (ça remplace `app/layout.js`, `app/page.js`, `app/globals.css`
générés par défaut — c'est voulu).

Copie aussi `tailwind.config.js` et `postcss.config.js` **seulement si
tu n'as pas déjà répondu "oui" à Tailwind CSS pendant `create-next-app`**
(dans ce cas ils existent déjà et sont probablement corrects).

## 2. Installer Tailwind si besoin

Si `tailwind.config.js` n'existait pas avant :

```bash
npm install -D tailwindcss postcss autoprefixer
```

## 3. Configurer l'URL du backend

Crée `.env.local` à la racine du frontend (copie `.env.local.example`) :

```env
NEXT_PUBLIC_API_URL=http://localhost:5000/api
```

## 4. Lancer les deux serveurs

Dans deux terminaux séparés :

```bash
# terminal 1 — backend
cd backend && npm run dev

# terminal 2 — frontend
cd frontend && npm run dev
```

Ouvre `http://localhost:3000`.

## 5. Ce que tu peux tester

- **/circuits** — liste, création, suppression
- **/circuits/[id]** — modification, suppression, liaison à des lieux et
  hôtels (menus déroulants), upload de photos/vidéos, suppression de
  médias
- **/tourist-places** — liste, création, modification, suppression
- **/hotels** — liste, création, modification, suppression

Pour tester les liaisons et l'upload, crée d'abord au moins un lieu
touristique et un hôtel (pages dédiées), puis va sur la fiche d'un
circuit pour les lui associer.

## Choix faits, à savoir

- **JavaScript, pas TypeScript** côté frontend — conforme à ta stack
  d'origine (le backend, lui, est en TypeScript).
- **`fetch` natif plutôt qu'Axios**, via `lib/api.js` — évite une
  dépendance en plus pour un besoin simple ; dis-moi si tu préfères
  Axios, c'est un changement mineur.
- **App Router** (pas de `react-router-dom` : Next.js gère le routing
  via la structure de dossiers `app/`).

## Ce qui n'est pas encore là (normal)

- Pas de connexion, pas de notion de "mon compte"
- Pas de sélecteur de langue (Phase 0 du planning, à faire séparément)
- Pas de carte, pas de réservation — ça vient aux phases suivantes

import AdminSidebar from "@/components/admin/AdminSidebar";

export default function AdminLayout({ children }) {
  const router = useRouter();
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function checkAccess() {
      const user = await getCurrentUser();

      if (!user) {
        router.replace("/login");
        return;
      }

      const role = await getCurrentRole();

      if (role !== "admin" && role !== "super_admin") {
        router.replace("/");
        return;
      }

      setLoading(false);
    }

    checkAccess();
  }, [router]);

  if (loading) {
    return (
      <main className="flex min-h-screen items-center justify-center bg-slate-950">
        <p className="text-sm text-slate-400">Vérification des accès...</p>

      <AdminSidebar />

      <main className="min-h-screen p-4 sm:p-6 lg:ml-64 lg:p-8">

  );

import { supabase } from "./supabase";

const API_URL =
  process.env.NEXT_PUBLIC_API_URL || "http://localhost:5000/api";

async function request(path, options = {}) {
  const isFormData = options.body instanceof FormData;

  const {
    data: { session },
  } = await supabase.auth.getSession();

  const headers = new Headers(options.headers || {});

  if (!isFormData) {
    headers.set("Content-Type", "application/json");
  }

  if (session?.access_token) {
    headers.set(
      "Authorization",
      `Bearer ${session.access_token}`
    );
  }

  const res = await fetch(`${API_URL}${path}`, {
    ...options,
    headers,
  });

  if (!res.ok) {
    let message = `Erreur ${res.status}`;

    try {
      const data = await res.json();
      message = data.message || message;
    } catch {
      // pas de corps JSON
    }

    throw new Error(message);
  }

  if (res.status === 204) return null;

  return res.json();
}

export const api = {
  get: (path) => request(path),

  post: (path, body) =>
    request(path, {
      method: "POST",
      body:
        body instanceof FormData
          ? body
          : JSON.stringify(body),
    }),

  put: (path, body) =>
    request(path, {
      method: "PUT",
      body: JSON.stringify(body),
    }),

  del: (path) =>
    request(path, {
      method: "DELETE",
    }),
};
